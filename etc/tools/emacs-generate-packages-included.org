#+NAME: list-all-packages
#+begin_src elisp
(require 'cl-lib)

(let ((packages '()))
  (dolist (file (directory-files "~/.config/emacs/site-lisp" t "\\.el"))
   (with-temp-buffer
    (insert-file-contents file)
    (goto-char (point-min))
    (condition-case nil
        (while t
          (let ((form (read (current-buffer))))
            (when (and (listp form)
                       (eq (car form) 'use-package))
              (push (symbol-name (cadr form)) packages))))
      (end-of-file))))
  (sort packages #'string<)
  (princ (concat ";; " (s-join ", " packages))))
#+end_src
